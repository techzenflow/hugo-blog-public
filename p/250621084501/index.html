* {
  margin: 0;
  padding: 0;
  box-sizing: border-box; /* 确保元素的宽度包含 padding 和 border */
}

ul, ol {
  list-style: none;
}

a {
  color: inherit;
  text-decoration: none;
}

input, button, textarea, select {
  font-family: inherit;
  font-size: inherit;
  border: none;
  background: none;
  outline: none;
}
button {
  cursor: pointer;
}

img {
  vertical-align: middle; /* 或 top/bottom */
  display: block; /* 可选，避免行内元素的间隙 */
}

table {
  border-collapse: collapse;
  border-spacing: 0;
}

h1, h2, h3, h4, h5, h6 {
  font-size: inherit;
  font-weight: normal;
}

body {
  font-family: 'Arial', sans-serif;
  line-height: 1.5;
}
